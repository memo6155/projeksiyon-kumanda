Android Studio Bluetooth HID Köprüsü

Bu arayüzü doğrudan APK içinde çalıştırın

Bu HTML dosyası her tuş basımında otomatik olarak window.AndroidBridge ve window.AndroidHidBridge metodlarını tetikler.

Android Studio'da bir WebView oluşturup aşağıdaki JavascriptInterface sınıfını bağlayarak telefonunuzu gerçek bir Bluetooth HID Kumandaya dönüştürebilirsiniz:

// Android Studio -> MainActivity.kt (BluetoothHidDevice Köprüsü)
class AndroidHidBridge(
    private val context: Context,
    private val hidDevice: BluetoothHidDevice?,
    private val hostDevice: BluetoothDevice?
) {
    // 1. Klavye (0x01) veya Multimedya Consumer (0x02) Tuşu Gönder
    @JavascriptInterface
    fun sendKey(reportId: Int, hexBytesJson: String, keyName: String) {
        val bytes = parseJsonBytes(hexBytesJson)
        hidDevice?.sendReport(hostDevice, reportId, bytes)
        // Tuş bırakma (Release) raporu
        val releaseBytes = ByteArray(bytes.size) { 0 }
        hidDevice?.sendReport(hostDevice, reportId, releaseBytes)
    }

    // 2. Sesli Yazılan Metni Karakter Karakter Gönder
    @JavascriptInterface
    fun sendText(text: String, asciiMode: Boolean) {
        // Her karakter için HID Keyboard (Report ID 1) gönderilir
    }

    // 3. Projeksiyon Touchpad Fare Hareketi Gönder (Report ID 3)
    @JavascriptInterface
    fun sendMouse(buttons: Int, dx: Int, dy: Int, wheel: Int) {
        val report = byteArrayOf(
            buttons.toByte(),
            dx.coerceIn(-127, 127).toByte(),
            dy.coerceIn(-127, 127).toByte(),
            wheel.coerceIn(-127, 127).toByte()
        )
        hidDevice?.sendReport(hostDevice, 3, report)
    }
}

// WebView Bağlantısı:
webView.settings.javaScriptEnabled = true
webView.addJavascriptInterface(AndroidHidBridge(this, hid, host), "AndroidBridge")
webView.loadUrl("file:///android_asset/index.html")
